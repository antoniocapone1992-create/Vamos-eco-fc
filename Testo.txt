        function getQuotaDovutaCorrente() {
            const now = new Date();
            const month = now.getMonth();
            let total = 0;
            if (month >= 8) {
                let months80 = Math.min(3, month - 8 + 1);
                total += months80 * 80;
                if (month >= 11) {
                    let months25 = month - 11 + 1;
                    total += months25 * 25;
                }
            } else {
                total += 3 * 80 + 1 * 25;
                let months25CurrentYear = month + 1;
                total += months25CurrentYear * 25;
            }
            return Math.min(QUOTA_TOTALE_ANNUA, total);
        }

        function updateCountdown() {
            const targetDate = new Date(2026, 10, 20, 21, 0, 0).getTime();
            const now = new Date().getTime();
            const diff = targetDate - now;

            const daysEl = document.getElementById('days');
            const hoursEl = document.getElementById('hours');
            const minutesEl = document.getElementById('minutes');
            const secondsEl = document.getElementById('seconds');

            if (!daysEl || !hoursEl || !minutesEl || !secondsEl) return;

            if (diff > 0) {
                daysEl.innerText = String(Math.floor(diff / (1000 * 60 * 60 * 24))).padStart(2, '0');
                hoursEl.innerText = String(Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60))).padStart(2, '0');
                minutesEl.innerText = String(Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60))).padStart(2, '0');
                secondsEl.innerText = String(Math.floor((diff % (1000 * 60)) / 1000)).padStart(2, '0');
            } else {
                daysEl.innerText = "00";
                hoursEl.innerText = "00";
                minutesEl.innerText = "00";
                secondsEl.innerText = "00";
            }
        }

        function saveAllRosterChanges() {
            let hasError = false;
            roster.forEach((p, globalIndex) => {
                const allInput = document.getElementById(`input-all-${globalIndex}`);
                const presInput = document.getElementById(`input-pres-${globalIndex}`);
                if (allInput && presInput) {
                    const newAll = parseInt(allInput.value);
                    const newPres = parseInt(presInput.value);
                    if (isNaN(newAll) || isNaN(newPres) || newAll < 0 || newPres < 0 || newPres > newAll) {
                        hasError = true;
                    } else {
                        roster[globalIndex].allenamenti = newAll;
                        roster[globalIndex].presenze = newPres;
                    }
                }
            });

            if (hasError) {
                alert('Errore: controlla che i valori inseriti siano numeri validi e che le presenze non superino gli allenamenti per nessun giocatore.');
                return;
            }

            window.dbSet(window.dbRef(window.db, 'vamos_roster'), roster).then(() => {
                alert('Tutti i dati della rosa sono stati aggiornati e sincronizzati sul cloud!');
                toggleRosaAdmin();
            }).catch((error) => {
                alert('Errore durante il salvataggio su Firebase: ' + error.message);
            });
        }

        function getSlotsForFormation(formation) {
            let lineupStruct = emptyTacticalLineup[formation] || emptyTacticalLineup["4-4-2"];
            let slots = [];
            for (let i = 0; i < lineupStruct.por.length; i++) {
                slots.push({ label: `Portiere`, role: "POR" });
            }
            for (let i = 0; i < lineupStruct.dif.length; i++) {
                slots.push({ label: `Difensore ${i+1}`, role: "DIF" });
            }
            for (let i = 0; i < lineupStruct.cen.length; i++) {
                slots.push({ label: `Centrocampista ${i+1}`, role: "CEN" });
            }
            for (let i = 0; i < lineupStruct.att.length; i++) {
                slots.push({ label: `Attaccante ${i+1}`, role: "ATT" });
            }
            return slots;
        }

        function changeVotingFormation(formVal) {
            selectedVotingFormation = formVal;
            const slotsHeader = document.querySelector('#voting-area h3');

        function renderVoteSelectors() {
            const selectorsDiv = document.getElementById('vote-selectors');
            if (!selectorsDiv) return;
            selectorsDiv.innerHTML = '';
            
            const slots = getSlotsForFormation(selectedVotingFormation);

            slots.forEach((slot, idx) => {
                const filteredPlayers = roster.filter(r => r.ruolo === slot.role);
                const div = document.createElement('div');
                div.innerHTML = `
                    <label class="block text-[11px] font-bold text-slate-400 uppercase mb-1">${slot.label} (${slot.role})</label>
                    <select id="vote-slot-${idx}" onchange="updateVotingDropdowns()" class="vote-dropdown w-full bg-slate-950 border border-slate-700 text-white text-xs rounded-xl p-2.5 outline-none">
                        <option value="">Seleziona giocatore...</option>
                        ${filteredPlayers.map(fp => `<option value="${fp.nome}">${fp.nome} (#${fp.n})</option>`).join('')}
                    </select>
                `;
                selectorsDiv.appendChild(div);
            });
            updateVotingDropdowns();
        }

        function updateVotingDropdowns() {
            const dropdowns = document.querySelectorAll('.vote-dropdown');
            const selectedValues = Array.from(dropdowns).map(dd => dd.value).filter(val => val !== "");

            dropdowns.forEach(dd => {
                const currentValue = dd.value;
                const options = dd.querySelectorAll('option');
                options.forEach(opt => {
                    if (opt.value === "") return;
                    if (selectedValues.includes(opt.value) && opt.value !== currentValue) {
                        opt.disabled = true;
                    } else {
                        opt.disabled = false;
                    }
                });
            });
        }

        function submitVote() {
            const dropdowns = document.querySelectorAll('.vote-dropdown');
            let selectedCount = 0;
            let currentSelections = [];

            dropdowns.forEach(dd => {
                const val = dd.value;
                if (val) {
                    selectedCount++;
                    currentSelections.push(val);
                }
            });

            if (selectedCount < 11) {
                alert('Seleziona tutti gli 11 titolari prima di inviare il voto.');
                return;
            }

            const uniqueSelections = new Set(currentSelections);
            if (uniqueSelections.size !== currentSelections.length) {
                alert('Attenzione: non puoi selezionare lo stesso giocatore più volte nella stessa formazione!');
                return;
            }

            if (!window.db) {
                alert('Configurazione Firebase mancante nel codice!');
                return;
            }

            const votesRef = window.dbRef(window.db, 'vamos_votes');
            window.dbGet(votesRef).then((snapshot) => {
                let existingVotes = snapshot.val() || {};
                currentSelections.forEach(playerName => {
                    existingVotes[playerName] = (existingVotes[playerName] || 0) + 1;
                });
                window.dbSet(votesRef, existingVotes).then(() => {
                    localStorage.setItem('vamos_voted_round', currentRound);
                    alert('Voto anonimo inviato al cloud con successo!');
                    initVotingSection();
                });
            });
        }

        function renderFinancesTable() {
            const tbody = document.getElementById('finanze-table-body');
            tbody.innerHTML = '';
            let totalCollected = 0;
            const quotaDovutaCorrente = getQuotaDovutaCorrente();

            document.getElementById('quota-info-text').innerText = `Quota totale annua: ${QUOTA_TOTALE_ANNUA}€ • Quota dovuta corrente (al periodo attuale): ${quotaDovutaCorrente}€`;

            roster.forEach(p => {
                const paid = finances[p.nome] || 0;
                totalCollected += paid;
                
                const isCompliant = paid >= quotaDovutaCorrente;
                let balanceHtml = "";

                if (isCompliant) {
                    let diff = paid - quotaDovutaCorrente;
                    balanceHtml = diff > 0 ? `+${diff} € (In regola)` : `In regola (0 €)`;
                } else {
                    let diff = quotaDovutaCorrente - paid;
                    balanceHtml = `-${diff} € (Mancanti)`;
                }

                const tr = document.createElement('tr');
                tr.className = "hover:bg-slate-800/40 transition-colors";
                tr.innerHTML = `
                    <td class="p-3.5 text-center font-bold text-amber-400">${p.n}</td>
                    <td class="p-3.5 font-medium text-white">${p.nome}</td>
                    <td class="p-3.5 text-center text-slate-300 font-bold">${quotaDovutaCorrente} €</td>
                    <td class="p-3.5 text-center">
                        <input type="number" value="${paid}" min="0" onchange="updatePaidAmount('${p.nome}', this.value)" class="w-24 bg-slate-950 border border-slate-700 text-white text-center text-xs rounded-lg p-1.5 outline-none focus:border-amber-500">
                    </td>
                    <td class="p-3.5 text-center ${isCompliant ? 'text-green-400 font-bold' : 'text-red-400 font-bold'}">${balanceHtml}</td>
                    <td class="p-3.5 text-center">
                        <span class="text-[10px] text-slate-400">${paid >= QUOTA_TOTALE_ANNUA ? 'Saldo Totale' : 'In corso'}</span>
                    </td>
                `;
                tbody.appendChild(tr);
            });

            document.getElementById('total-collected').innerText = totalCollected + ' €';
        }

        function updatePaidAmount(playerName, value) {
            const val = parseFloat(value);
            if (!isNaN(val) && val >= 0) {
                finances[playerName] = val;
                window.dbSet(window.dbRef(window.db, 'vamos_finances'), finances).then(() => {
                    renderFinancesTable();
                });
            } else {
                alert('Inserisci un importo valido.');
                renderFinancesTable();
            }
        }
